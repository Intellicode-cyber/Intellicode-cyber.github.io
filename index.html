analyse_notes.py
def analyser(notes):
    moy = sum(notes) / len(notes)
    sup = [n for n in notes if n >= moy]
    return round(moy, 1), max(notes), sup

notes = [72, 88, 95, 64, 81]
moy, top, sup = analyser(notes)
print(f"Moyenne : {moy} | Max : {top}")
print("Au-dessus :", sup)
# Moyenne : 80.0 | Max : 95
# Au-dessus : [88, 95, 81]
